Simpan otomatis data tabel ke Sheet & file PDF Surat Tanah ke Google Drive

Saat Admin menyimpan aset dengan dokumen PDF, file PDF otomatis dibuat di folder Google Drive di atas dan tautan Drive-nya tercatat di Google Sheet.

Kode Google Apps Script (Sheet + Auto Upload PDF ke Drive):
  1. Buka Google Sheet baru, klik menu Ekstensi > Apps Script.
  2. Tempel kode di bawah ini, lalu klik Terapkan (Deploy) > Deployment baru > Aplikasi Web.
  3. Pilih Akses: Siapa saja (Anyone), izinkan akses Drive & Sheet, lalu salin URL Web App ke atas.
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  var driveUrl = data.pdfDriveUrl || "";

  if (data.pdfBase64 && data.pdfFileName) {
    var folderName = data.driveFolderName || "Dokumen_Aset_Pesona_Kahuripan";
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
    var decoded = Utilities.base64Decode(data.pdfBase64);
    var blob = Utilities.newBlob(decoded, "application/pdf", data.code + "_" + data.pdfFileName);
    var file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    driveUrl = file.getUrl();
  }

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Waktu Sync", "Kode Aset", "Atas Nama", "Kategori", "Status Legalitas", "No. Sertifikat", "Luas (m2)", "Status Aset", "Alamat", "Koordinat Pusat", "Tanggal Input", "Dokumen PDF", "Link Dokumen PDF (Google Drive)", "Keterangan", "Koordinat Poligon"]);
  }
  sheet.appendRow([new Date(), data.code, data.name, data.category, data.legality, data.certNo, data.area, data.status, data.location, data.center, data.date, data.pdfFileName || "-", driveUrl, data.notes, JSON.stringify(data.polygon || [])]);
  return ContentService.createTextOutput(JSON.stringify({status: "success", driveUrl: driveUrl})).setMimeType(ContentService.MimeType.JSON);
}
